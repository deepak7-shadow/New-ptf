// Smart Pesticide Spraying Nozzle - Sensor Loop
#include <WiFi.h>
#include <BlynkSimpleEsp32.h>

const int NOZZLE_PIN = 23;
const int SENSOR_PIN = 34;

void setup() {
  Serial.begin(115200);
  pinMode(NOZZLE_PIN, OUTPUT);
  Blynk.begin(AUTH_TOKEN, SSID, PASS);
  Serial.println("[SYSTEM READY: SENSE · DECIDE · SPRAY]");
}

void loop() {
  Blynk.run();
  float cropCondition = analogRead(SENSOR_PIN);
  if (cropCondition > THRESHOLD_INFESTATION) {
    digitalWrite(NOZZLE_PIN, HIGH); // Targeted Precision Spray
    Blynk.virtualWrite(V1, "SPRAY_ACTIVE");
  } else {
    digitalWrite(NOZZLE_PIN, LOW);  // Preserve Chemical
  }
}
